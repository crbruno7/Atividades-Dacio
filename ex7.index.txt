<script>
    function validateLuhn(number) {
        let sum = 0;
        let shouldDouble = false;
        
        // Loop de trás para frente
        for (let i = number.length - 1; i >= 0; i--) {
            let digit = parseInt(number.charAt(i));

            if (shouldDouble) {
                digit *= 2;
                if (digit > 9) digit -= 9;
            }

            sum += digit;
            shouldDouble = !shouldDouble;
        }
        return (sum % 10) === 0;
    }

    function getCardDetails(number) {
        const mii = number[0];
        const sectors = {
            '1': 'Companhias Aéreas',
            '2': 'Companhias Aéreas / Outros',
            '3': 'Viagens e Entretenimento (American Express/Diners)',
            '4': 'Serviços Bancários e Financeiros (Visa)',
            '5': 'Serviços Bancários e Financeiros (Mastercard)',
            '6': 'Merchandising e Bancário (Discover/Maestro)',
            '7': 'Petróleo',
            '8': 'Telecomunicações',
            '9': 'Atribuição Nacional'
        };

        let brand = "Desconhecida";
        if (/^4/.test(number)) brand = "Visa";
        else if (/^5[1-5]/.test(number) || /^2[2-7]/.test(number)) brand = "Mastercard";
        else if (/^3[47]/.test(number)) brand = "American Express";
        else if (/^6(?:011|5)/.test(number)) brand = "Discover";
        else if (/^3(?:0[0-5]|[68])/.test(number)) brand = "Diners Club";
        else if (/^(?:2131|1800|35)/.test(number)) brand = "JCB";
        else if (/^6062/.test(number)) brand = "Hipercard";

        return {
            brand: brand,
            sector: sectors[mii] || "Desconhecido",
            mii: mii
        };
    }

    function analyzeCard() {
        const rawValue = document.getElementById('cardNumber').value;
        const cleaned = rawValue.replace(/[\s.-]/g, '');
        const panel = document.getElementById('panel');
        
        if (cleaned.length < 13 || cleaned.length > 16 || isNaN(cleaned)) {
            alert("Por favor, insira um número entre 13 e 16 dígitos.");
            return;
        }

        const isValid = validateLuhn(cleaned);
        const details = getCardDetails(cleaned);

        // Preencher Painel
        const resStatus = document.getElementById('resStatus');
        resStatus.innerText = isValid ? "VÁLIDO" : "INVÁLIDO (Luhn Falhou)";
        resStatus.className = "info-value " + (isValid ? "status-valid" : "status-invalid");

        document.getElementById('resBrand').innerText = details.brand;
        document.getElementById('resSector').innerText = details.sector;
        document.getElementById('resMii').innerText = details.mii;

        panel.style.display = 'block';
    }
</script>
